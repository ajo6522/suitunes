cmake_minimum_required(VERSION 3.24)
project(Suitunes VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Suitunes Windows builds require x64")
endif()
include(FetchContent)
set(SUITUNES_JUCE_DIR "" CACHE PATH "Optional existing JUCE 8.0.8 source")
if(SUITUNES_JUCE_DIR)
  add_subdirectory("${SUITUNES_JUCE_DIR}" JUCE)
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
add_subdirectory(vendor/signalsmith-stretch)
set(SUITUNES_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND SUITUNES_FORMATS AU)
endif()
juce_add_plugin(Suitunes COMPANY_NAME "2of3" BUNDLE_ID com.twoofthree.suitunes
  PLUGIN_MANUFACTURER_CODE Tof3 PLUGIN_CODE SuTn PRODUCT_NAME Suitunes
  FORMATS ${SUITUNES_FORMATS} IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx PitchShift)
target_sources(Suitunes PRIVATE Source/Plugin.cpp Source/Engine.h Source/Preset.h)
target_include_directories(Suitunes PRIVATE Source)
target_compile_options(Suitunes PUBLIC $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_compile_definitions(Suitunes PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Suitunes PRIVATE juce::juce_audio_utils juce::juce_dsp signalsmith-stretch
  PUBLIC juce::juce_recommended_config_flags)
include(CTest)
if(BUILD_TESTING)
  juce_add_console_app(SuitunesTests PRODUCT_NAME "Suitunes tests")
  target_sources(SuitunesTests PRIVATE Tests/Tests.cpp)
  target_include_directories(SuitunesTests PRIVATE Source)
  target_compile_definitions(SuitunesTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_compile_definitions(SuitunesTests PRIVATE "SUITUNES_FIXTURE_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/Tests/fixtures\"")
  target_link_libraries(SuitunesTests PRIVATE Suitunes juce::juce_audio_utils juce::juce_dsp signalsmith-stretch)
  add_test(NAME SuitunesDSPAndPresets COMMAND SuitunesTests)
  add_executable(SuitunesBench Tests/Benchmark.cpp)
  target_include_directories(SuitunesBench PRIVATE Source)
  target_link_libraries(SuitunesBench PRIVATE juce::juce_audio_basics signalsmith-stretch)
endif()
